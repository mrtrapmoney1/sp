  • ${idx + 1} ${item.part} ${item.count}
    • `).join(''); } else { partsList.innerHTML = '
  • No data available
    • '; } // Render common problems const problemsList = document.getElementById('top-problems-list'); if (analytics.common_problems && analytics.common_problems.length > 0) { problemsList.innerHTML = analytics.common_problems.slice(0, 10).map((item, idx) => `
  • ${idx + 1} ${truncate(item.problem, 40)} ${item.count}
    • `).join(''); } else { problemsList.innerHTML = '
  • No data available
    • '; } } function renderCustomerAnalytics(analytics) { // Update summary document.getElementById('unique-cities').textContent = analytics.unique_cities || 0; // Render brands chart if (analytics.top_makes && analytics.top_makes.length > 0) { const labels = analytics.top_makes.map(m => m.make); const data = analytics.top_makes.map(m => m.count); if (brandsChart) brandsChart.destroy(); brandsChart = new Chart(document.getElementById('brands-chart'), { type: 'bar', data: { labels: labels, datasets: [{ label: 'Service Calls', data: data, backgroundColor: 'rgba(26, 54, 93, 0.8)', borderColor: 'rgba(26, 54, 93, 1)', borderWidth: 1 }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true } } } }); } // Render products chart if (analytics.top_product_types && analytics.top_product_types.length > 0) { const labels = analytics.top_product_types.map(p => p.type); const data = analytics.top_product_types.map(p => p.count); const colors = [ '#1a365d', '#2c5282', '#3182ce', '#4299e1', '#63b3ed', '#48bb78', '#ed8936', '#fc8181', '#9f7aea', '#667eea' ]; if (productsChart) productsChart.destroy(); productsChart = new Chart(document.getElementById('products-chart'), { type: 'doughnut', data: { labels: labels, datasets: [{ data: data, backgroundColor: colors.slice(0, data.length) }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'right' } } } }); } // Render top cities const citiesList = document.getElementById('top-cities-list'); if (analytics.top_cities && analytics.top_cities.length > 0) { citiesList.innerHTML = analytics.top_cities.slice(0, 10).map((item, idx) => `
  • ${idx + 1} ${item.city} ${item.count}
    • `).join(''); } else { citiesList.innerHTML = '
  • No data available
    • '; } } function truncate(str, len) { if (!str) return ''; return str.length > len ? str.substring(0, len) + '...' : str; } // Load analytics on page load loadAnalytics(); {% endblock %}
