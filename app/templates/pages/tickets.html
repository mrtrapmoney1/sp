// Ticket Lookup Function
async function lookupTicket(event) {
    event.preventDefault();

    const reference = document.getElementById('lookup-reference').value.trim();
    if (!reference) {
        showAlert('Please enter a search term', 'warning');
        return;
    }

    const btn = document.getElementById('lookup-btn');
    btn.disabled = true;
    btn.innerHTML = ' Searching...';

    try {
        const response = await fetch('/api/tickets/lookup', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ reference: reference })
        });

        const data = await response.json();

        if (data.success) {
            displaySearchResults(data.tickets, data.count);
            if (data.count > 0) {
                showAlert(`Found ${data.count} ticket(s)`, 'success');
            } else {
                showAlert('No tickets found matching your search', 'info');
            }
        } else {
            showAlert(data.error || 'Search failed', 'danger');
        }
    } catch (err) {
        console.error('Lookup error:', err);
        showAlert('Error searching tickets: ' + err.message, 'danger');
    } finally {
        btn.disabled = false;
        btn.innerHTML = `
            
            Search
        `;
    }
}

// Load ticket data into form
function loadTicket(index) {
    const ticket = window.searchResults[index];
    if (!ticket) return;

    // Mark row as selected
    selectTicket(index);

    // Populate form fields
    const fieldMappings = {
        'invoice': 'INVOICE',
        'lastname': 'LASTNAME',
        'firstname': 'FIRSTNAME',
        'address': 'ADDRESS',
        'city': 'CITY',
        'state': 'STATE',
        'zip': 'ZIP',
        'phone': 'PHONE',
        'phone2': 'PHONE2',
        'make': 'MAKE',
        'typ': 'TYP',
        'model': 'MODEL',
        'serial': 'SERIAL',
        'servicereq': 'SERVICEREQ',
        'location': 'LOCATION',
        'accessor': 'ACCESSOR',
        'dlrinvoice': 'DLRINVOICE',
        'btaddress': 'BTADDRESS'
    };

    for (const [formField, dbField] of Object.entries(fieldMappings)) {
        const element = document.getElementById(formField);
        if (element && ticket[dbField]) {
            element.value = ticket[dbField];
        }
    }

    // Handle date fields
    if (ticket.DATEIN) {
        const datein = parseDate(ticket.DATEIN);
        if (datein) document.getElementById('datein').value = datein;
    }
    if (ticket.DATEPROM) {
        const dateprom = parseDate(ticket.DATEPROM);
        if (dateprom) document.getElementById('dateprom').value = dateprom;
    }
    if (ticket.DATEPUR) {
        const datepur = parseDate(ticket.DATEPUR);
        if (datepur) document.getElementById('datepur').value = datepur;
    }

    // Update UI to show loaded ticket
    loadedTicketId = ticket.INVOICE || ticket.DLRINVOICE || 'Unknown';
    document.getElementById('form-title').textContent = 'Edit Ticket';
    document.getElementById('loaded-ticket-badge').style.display = 'inline-flex';
    document.getElementById('loaded-ticket-id').textContent = loadedTicketId;
    document.getElementById('clear-btn').style.display = 'inline-flex';

    // Scroll to form
    document.getElementById('ticket-form').scrollIntoView({ behavior: 'smooth', block: 'start' });

    showAlert(`Loaded ticket ${loadedTicketId}`, 'success');
}

// Parse date from various formats
function parseDate(dateValue) {
    if (!dateValue) return null;

    // Handle Date objects
    if (dateValue instanceof Date) {
        return dateValue.toISOString().split('T')[0];
    }

    // Handle string formats
    const str = String(dateValue).trim();
    if (!str || str === 'None' || str === 'nan') return null;

    // Try to parse
    try {
        const date = new Date(str);
        if (!isNaN(date.getTime())) {
            return date.toISOString().split('T')[0];
        }
    } catch {
        // Ignore parse errors
    }

    return null;
}

// Format date for display
function formatDate(dateValue) {
    const parsed = parseDate(dateValue);
    if (!parsed) return '';
    try {
        return new Date(parsed).toLocaleDateString();
    } catch {
        return '';
    }
}

// Form submission
document.getElementById('ticket-form').addEventListener('submit', async function(e) {
    e.preventDefault();

    const formData = new FormData(this);
    const ticket = Object.fromEntries(formData.entries());

    if (!ticket.lastname) {
        showAlert('Last name is required', 'warning');
        return;
    }

    try {
        const response = await fetch('/api/tickets', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(ticket)
        });

        const data = await response.json();

        if (data.success) {
            showAlert('Ticket saved successfully!', 'success');
            if (!loadedTicketId) {
                // Only reset if creating new ticket
                this.reset();
            }
        } else {
            showAlert(data.error || 'Failed to save ticket', 'danger');
        }
    } catch (err) {
        showAlert('Error: ' + err.message, 'danger');
    }
});

async function exportTicket() {
    const form = document.getElementById('ticket-form');
    const formData = new FormData(form);
    const ticket = Object.fromEntries(formData.entries());

    if (!ticket.lastname) {
        showAlert('Please enter at least a last name', 'warning');
        return;
    }

    try {
        const response = await fetch('/api/export/dbf', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                calls: [{
                    ...ticket,
                    invoice: ticket.invoice || Date.now().toString().slice(-5)
                }]
            })
        });

        if (response.ok) {
            const blob = await response.blob();
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `ticket_${ticket.invoice || Date.now()}.dbf`;
            document.body.appendChild(a);
            a.click();
            window.URL.revokeObjectURL(url);
            a.remove();
            showAlert('DBF file downloaded', 'success');
        } else {
            const data = await response.json();
            showAlert(data.error || 'Export failed', 'danger');
        }
    } catch (err) {
        showAlert('Error: ' + err.message, 'danger');
    }
}

function showAlert(message, type) {
    const container = document.getElementById('alert-container');
    const alertId = 'alert-' + Date.now();
    const alertHtml = `
        

            ${message}
        

    `;
    container.insertAdjacentHTML('beforeend', alertHtml);

    if (type !== 'danger') {
        setTimeout(() => {
            const alert = document.getElementById(alertId);
            if (alert) alert.remove();
        }, 4000);
    }
}
