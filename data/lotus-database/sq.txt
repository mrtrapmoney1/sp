DROP TABLE IF EXISTS Cust;
DROP TABLE IF EXISTS CityStZip;
DROP TABLE IF EXISTS Ticket;
DROP TABLE IF EXISTS Dispatch;
DROP TABLE IF EXISTS Dispatch_Company;
DROP TABLE IF EXISTS Item;
DROP TABLE IF EXISTS CoverageID;
DROP TABLE IF EXISTS ItemType;
DROP TABLE IF EXISTS TicLoc;
DROP TABLE IF EXISTS Phone;


CREATE TABLE Cust (
id INTEGER PRIMARY KEY NOT NULL DEFAULT 0,
LName VARCHAR(50) NOT NULL DEFAULT 'LName',
FName VARCHAR(50) NOT NULL DEFAULT 'Fname',
Address VARCHAR NOT NULL,
zipcode INTEGER NOT NULL,
PhoneID INTEGER NOT NULL,
FOREIGN KEY(zipcode) REFERENCES Cust(id),
FOREIGN KEY(PhoneID) REFERENCES Phone(PhoneID));

CREATE TABLE CityStZip (
City TEXT(50) NOT NULL,
State TEXT(2) NOT NULL DEFAULT NE,
Zip INTEGER PRIMARY KEY NOT NULL UNIQUE,
FOREIGN KEY(Zip) REFERENCES Cust(zipcode));

CREATE TABLE Ticket (
TicketID INTEGER(6) PRIMARY KEY AUTOINCREMENT NOT NULL UNIQUE,
CustID INTEGER(6) NOT NULL,
ItemID INTEGER(6) NOT NULL,
DispatchID INTEGER(6),
TicLocID INTEGER(2) NOT NULL,
FOREIGN KEY(CustID) REFERENCES Cust(id),
FOREIGN KEY(ItemID) REFERENCES Item(ItemId),
FOREIGN KEY(DispatchID) REFERENCES Dispatch(DispatchID),
FOREIGN KEY(TicLocID) REFERENCES TicLoc(LocationID));

CREATE TABLE Dispatch (
DispatchID INTEGER(6) PRIMARY KEY NOT NULL,
CompanyID INTEGER(6) NOT NULL,
Prescheduled_Date DATE,
Preapproved_amount NUMERIC(4,2) NOT NULL,
CoverageID INTEGER(3) NOT NULL,
FOREIGN KEY(CompanyID) REFERENCES Dispatch_Company(Company_ID),
FOREIGN KEY(CoverageID) REFERENCES CoverageID(CoverageID));

CREATE TABLE Dispatch_Company (
Company_ID INTEGER PRIMARY KEY NOT NULL,
Company_name TEXT(50) NOT NULL UNIQUE,
ManagerID INTEGER NOT NULL,
FOREIGN KEY(ManagerID) REFERENCES Cust(id));
CREATE TABLE Item (
ItemId INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL UNIQUE,
Model_Number VARCHAR(40),
Serial_Number VARCHAR(40) UNIQUE,
TypeID INTEGER(3) NOT NULL,
Color VARCHAR(20) NOT NULL,
Item_Location VARCHAR(30) NOT NULL,
Purchase_Date DATE(8),
Purchased_from VARCHAR(30) NOT NULL,
Purchased_City TEXT(30),
FOREIGN KEY(TypeID) REFERENCES ItemType(TypeID));

CREATE TABLE CoverageID (
Type VARCHAR(30) NOT NULL UNIQUE,
CoverageID INTEGER PRIMARY KEY NOT NULL);

CREATE TABLE ItemType (
TypeID INTEGER PRIMARY KEY NOT NULL UNIQUE,
Type Description TEXT(30) NOT NULL UNIQUE);

CREATE TABLE TicLoc (
LocationID INTEGER(2) PRIMARY KEY NOT NULL,
Location_Description TEXT(30) NOT NULL UNIQUE);

CREATE TABLE Phone (
PhoneID INTEGER PRIMARY KEY NOT NULL UNIQUE,
Phone_Type VARCHAR(15) NOT NULL,
Phone_Number NUMERIC(10) NOT NULL UNIQUE);
