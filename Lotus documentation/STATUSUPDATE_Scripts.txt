Sub connecttodb
	Dim c As New connection, q As New query, r As New resultset, tbl As table, tfn As String, tn As String, rec As Integer
	
	c.connectto("dbase iv")
	Set q.connection = c
	Set tbl = currentdocument.tables(0)
	
	tfn = tbl.fullname
	fn=tbl.tablename
	
	If Instr(1,tn,":") >0 Then
		tn = Left(tn,Len(tn)-2)
	End If
	
	q.sql = "select * from """+tfn+""" "+tn+""
	
	Set r.query = q
	
	If (r.execute) Then
		rec=r.numrows
		
	End If
	
	
End Sub

Sub OpnPymt
	Dim MyDoc As String
	Dim NumDocs As Integer
	Dim Docs As Variant
	Dim MyOpen As String
	Dim fname As String
	If(CurrentApplication.Documents.IsEmpty=False) Then
		NumDocs = CurrentApplication.Documents.count
		For i = 0 To NumDocs-1
			Print CurrentApplication.Documents(i).Name
			If CurrentApplication.Documents(i).Name="Payments" Then
				MyOpen="True"
			End If
		Next
	End If
	If MyOpen ="True" Then
		CurrentApplication.Payments.activate
	Else
		Set CurrentDocument= CurrentApplication.OpenDocument("Payments.apr","y:\Lotus")
	End If
End Sub

Sub SetDates
	If Day(Today) >= 15 Then
		StartDate = Datenumber(Year(Today),Month(Today),15)
		[VarStDate].Text = Cstr(StartDate)
		NextMo = Datenumber(Year(Today), Month(Today)+1,1)                           
		EndDate = Datenumber(Year(NextMo), Month(NextMo),Day(NextMo)-1)
		[VarEndDate].Text = Cstr(EndDate)
		CurrentWindow.Repaint
	Else
		StartDate = Datenumber(Year(Today),Month(Today),1)
		[VarStDate].Text = Cstr(StartDate)
		EndDate = Datenumber(Year(Today),Month(Today),14)
		[VarEndDate].Text = Cstr(EndDate)
		CurrentWindow.Repaint
	End If
End Sub
